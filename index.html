// Función dedicada exclusivamente a consultar y actualizar las cargas entregadas async function fetchDeliveries() { const container = document.getElementById('deliveries-container'); if (!container) return; try { // Consulta al endpoint API de Vercel const response = await fetch('/api/deliveries'); if (!response.ok) { throw new Error('No se pudo conectar con el endpoint de cargas'); } const data = await response.json(); const deliveries = data.deliveries || []; if (deliveries.length === 0) { container.innerHTML = `

Esperando nuevos registros o mensajes de prueba...

Los datos se actualizarán automáticamente cuando se envíe una carga o mensaje.

`; return; } // Renderizar cada registro container.innerHTML = deliveries.map(item => `

${escapeHtml(item.driver)}

${escapeHtml(item.timestamp || 'Ahora')}
${escapeHtml(item.status || 'En Tránsito')}
${item.description ? `

${escapeHtml(item.description)}

` : ''}
Carga / Detalle ${escapeHtml(item.cargo)}
Ruta ${escapeHtml(item.origin)} ➔ ${escapeHtml(item.destination)}
`).join(''); } catch (error) { console.error('Error fetching deliveries:', error); } } // Helper para evitar inyección de código function escapeHtml(str) { if (!str) return ''; return String(str) .replace(/&/g, "&") .replace(//g, ">") .replace(/"/g, """) .replace(/'/g, "'"); } // Ejecutar al cargar la página y refrescar cada 5 segundos document.addEventListener('DOMContentLoaded', () => { fetchDeliveries(); setInterval(fetchDeliveries, 5000); });
